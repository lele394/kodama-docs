def record_path(data_dir: str, relpath: str, kind: str, name: str,
                line: int) -> str:
    sub = _SUBDIR[kind]
    fhash = hashlib.md5(relpath.encode('utf-8')).hexdigest()[:10]
    fname = f'{kind}_{fhash}_{sanitize_id(name)}_{line}.json'
    return os.path.join(data_dir, sub, fname)

def save_record(data_dir: str, relpath: str, kind: str, name: str, line: int,
                prose: dict, fingerprint: str, symbol: dict,
                stubbed: bool = False):
    """Save one record. `stubbed=True` marks it as `--no-llm` placeholder
    prose: full-mode runs treat such records as cache misses (and overwrite
    them), --no-llm runs may reuse them.

    A stub never overwrites real prose. ``--no-llm`` is a diagnostic mode;
    losing paid/generated records because the prompt or a setting changed
    would be surprising and irreversible."""
    path = record_path(data_dir, relpath, kind, name, line)
    if stubbed:
        existing = _read_record(path)
        if existing is not None and not existing.get('stubbed'):
            return path
    os.makedirs(os.path.dirname(path), exist_ok=True)
    rec = {
        'schema': 'kodama_docs/record@1',
        'kind': kind,
        'file': relpath,
        'name': name,
        'line': line,
        'symbol': symbol,
        'prose': prose,
        'fingerprint': fingerprint,
    }
    if stubbed:
        rec['stubbed'] = True
    tmp = path + '.tmp'
    with open(tmp, 'w', encoding='utf-8') as fh:
        json.dump(rec, fh, ensure_ascii=False, indent=1)
    os.replace(tmp, path)
    return path

def load_record(data_dir: str, relpath: str, kind: str, name: str,
                line: int, fingerprint: str, accept_stub: bool = False):
    """Load a cached record's prose, or None on miss. Records flagged
    `stubbed` (i.e. `--no-llm` placeholders) count as a miss unless
    `accept_stub=True`, so a full run regenerates real prose over them."""
    path = record_path(data_dir, relpath, kind, name, line)
    if not os.path.exists(path):
        return None
    try:
        with open(path, 'r', encoding='utf-8') as fh:
            rec = json.load(fh)
    except (ValueError, OSError):
        return None
    if rec.get('fingerprint') != fingerprint:
        return None
    if rec.get('stubbed') and not accept_stub:
        return None
    return rec.get('prose')

def list_records(data_dir: str, kind: str):
    sub = _SUBDIR[kind]
    d = os.path.join(data_dir, sub)
    out = {}
    if not os.path.isdir(d):
        return out
    for fname in sorted(os.listdir(d)):
        if not fname.endswith('.json'):
            continue
        try:
            with open(os.path.join(d, fname), 'r', encoding='utf-8') as fh:
                rec = json.load(fh)
            out[(rec.get('file'), rec.get('name'), rec.get('line'))] = rec
        except (ValueError, OSError):
            continue
    return out

def delete_records_for(data_dir: str, relpaths) -> int:
    """Delete cached LLM records that belong to the given source files."""
    removed = 0
    for kind in _SUBDIR:
        for (file, name, line), _rec in list_records(data_dir, kind).items():
            if file in relpaths:
                try:
                    os.remove(record_path(data_dir, file, kind, name, line))
                    removed += 1
                except OSError:
                    pass
    return removed
