def _tree(base, files):
    for rel, content in files.items():
        p = os.path.join(base, rel)
        os.makedirs(os.path.dirname(p) or base, exist_ok=True)
        with open(p, 'w', encoding='utf-8') as fh:
            fh.write(content)

def test_list_marks_only_md_sorted():
    with tempfile.TemporaryDirectory() as d:
        _tree(d, {
            'zz.md': '# Zeta\n',
            'alpha.md': 'no heading here\n',
            'guide/setup.md': '# Setup\n',
            'code.lua': 'print(1)',
            'assets/style.css': 'body {}',
        })
        pages = manual.list_manual_pages(d)
        assert [p['rel'] for p in pages] == ['alpha.md', 'guide/setup.md',
                                             'zz.md'], [p['rel'] for p in pages]
        titles = {p['rel']: p['title'] for p in pages}
        assert titles['zz.md'] == 'Zeta'
        assert titles['alpha.md'] == 'alpha'      # falls back to the basename
        assert titles['guide/setup.md'] == 'Setup'
        dirs = {p['rel']: p['dir'] for p in pages}
        assert dirs['alpha.md'] == ''
        assert dirs['guide/setup.md'] == 'guide'

def test_copy_is_fresh_each_run():
    """A file removed from the source folder must not survive a re-copy."""
    with tempfile.TemporaryDirectory() as src, \
            tempfile.TemporaryDirectory() as out:
        _tree(src, {'keep.md': '# K\n', 'gone.md': '# G\n'})
        manual.copy_manual_docs(src, out)
        os.remove(os.path.join(src, 'gone.md'))
        pages = manual.copy_manual_docs(src, out)
        assert [p['rel'] for p in pages] == ['keep.md']
        assert not os.path.exists(
            os.path.join(out, manual.MANUAL_DIRNAME, 'gone.md'))

def test_index_section0_above_generated():
    finfos = [{'relpath': 'src/main.lua', 'lines': 10,
               'functions': [{'name': 'f'}], 'file_vars': []}]
    pages = [{'rel': 'intro.md', 'title': 'Intro', 'dir': ''}]
    md = render_index(finfos, '/out', manual_pages=pages)
    i_manual = md.find('## Manual documentation')
    i_gen = md.find('files/src/main.lua.md')
    assert i_manual != -1, 'manual section missing'
    assert i_gen != -1, 'generated table missing'
    assert i_manual < i_gen, 'manual section must come before the generated table'
    assert 'manual/intro.md' in md

    assert '## Manual documentation' not in render_index(finfos, '/out')

def test_index_manual_nested_bullets():
    """Section 0 is a nested bullet list mirroring the folder hierarchy:
    folders first (plain bullets), then the pages (links), at every depth."""
    finfos = [{'relpath': 'src/main.lua', 'lines': 10,
               'functions': [], 'file_vars': []}]
    pages = [
        {'rel': 'root.md', 'title': 'Root', 'dir': ''},
        {'rel': 'a/one.md', 'title': 'One', 'dir': 'a'},
        {'rel': 'a/two.md', 'title': 'Two', 'dir': 'a'},
        {'rel': 'a/b/deep.md', 'title': 'Deep', 'dir': 'a/b'},
    ]
    md = render_index(finfos, '/out', manual_pages=pages)
    sec = md.split('## Manual documentation')[1].split('\n\n1 files')[0]
    lines = [l for l in sec.splitlines() if l.strip()]
    assert lines == [
        '- a',
        '  - b',
        '    - [Deep](manual/a/b/deep.md)',
        '  - [One](manual/a/one.md)',
        '  - [Two](manual/a/two.md)',
        '- [Root](manual/root.md)',
    ], lines
    # the old flat table is gone
    assert '| Page |' not in md
