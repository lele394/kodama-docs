def _tree(base, files):
    for rel, content in files.items():
        p = os.path.join(base, rel)
        os.makedirs(os.path.dirname(p) or base, exist_ok=True)
        with open(p, 'wb' if isinstance(content, bytes) else 'w',
                  encoding=None if isinstance(content, bytes) else 'utf-8') as fh:
            fh.write(content)

def _inline_json_in(html: str, marker: str):
    """Parse the JSON inlined after ``marker`` in a rendered page: it runs
    from the marker to the first literal ``</script>`` (the builder escapes
    ``</`` in the data to ``<\\/``, so nothing inside can terminate the tag
    early and the first real ``</script>`` is the tag's own end)."""
    i = html.index(marker) + len(marker)
    return json.loads(html[i:html.index('</script>', i)])

def test_rewrite_md_links():
    rw = static_site.rewrite_md_links
    check(rw('<a href="x.md">') == '<a href="x.html">', 'plain .md -> .html')
    check(rw('<a href="x.md#a">') == '<a href="x.html#a">', 'fragment kept')
    check(rw('<a href="../y.md#b">') == '<a href="../y.html#b">',
          'relative link + fragment')
    ext = '<a href="https://e.com/x.md">'
    check(rw(ext) == ext, 'https link untouched')
    rel = '<a href="//cdn/x.md">'
    check(rw(rel) == rel, 'protocol-relative link untouched')
    check(rw('<a href="#a">') == '<a href="#a">', 'anchor untouched')
    check(rw('<a href="g/01%20-%20P.md#c">')
          == '<a href="g/01%20-%20P.html#c">', '%20-encoded link rewritten')

def test_diet_graph():
    g = gmod.nx.MultiDiGraph()
    g.add_node('fn:a.py::f@1', kind='function', relpath='a.py', name='f',
               line=1, end_line=9, params=['x'], vararg=False,
               body='SECRET-SOURCE', file_id='file:a.py')
    g.add_node('file:a.py', kind='file', relpath='a.py')
    g.add_edge('file:a.py', 'fn:a.py::f@1', type='CONTAINS')
    full = gmod.nx.node_link_data(g, edges='edges')
    diet = gmod.diet_graph_data(g)
    check(any(n.get('body') for n in full['nodes']),
          'diet: full graph keeps bodies')
    check(all('body' not in n for n in diet['nodes']),
          'diet: drops source bodies')
    check(all('params' not in n and 'end_line' not in n
              for n in diet['nodes']), 'diet: drops pipeline-only fields')
    check(diet['edges'] == full['edges'], 'diet: keeps all relationships')

    with tempfile.TemporaryDirectory() as tmp:
        docs, webdir = build_fixture(tmp)
        cfg = {'output_dir': docs, 'data_dir': os.path.join(docs, 'data')}

        site = static_site.build(cfg, webdir=webdir)
        check(site == os.path.join(docs, 'site'), 'site lives in <out>/site')

        # ---- layout ------------------------------------------------------
        for rel in (
                'index.html', 'graph.html', '404.html', '.nojekyll',
                'files/src/a.html', 'files/b.html',
                'manual/g/01 - Page.html', 'manual/img/pic.png',
                'app.js', 'sidebar.js', 'graph.js',
                'lib/hi.css',
                'theme/dark/style.css', 'theme/light/style.css',
                'symbols.json', 'symbols-data.js', 'graph.json',
        ):
            check(os.path.isfile(os.path.join(site, rel)), f'present: {rel}')
        # the module sub-files ship INSIDE the bundles (file:// cannot load
        # external ES modules), so the raw copies are absent
        for rel in ('fuzzy.js', 'graph/state.js'):
            check(not os.path.exists(os.path.join(site, rel)),
                  f'absent: {rel} (inlined into the bundle)')
        check(not os.path.isdir(os.path.join(site, 'graph')),
              'absent: graph/ dir (inlined into the bundle)')

        # ---- theme (dark, from the fixture web_settings.yaml) -----------
        idx = _read(os.path.join(site, 'index.html'))
        check('theme/dark/style.css' in idx, 'index: dark theme baked in')
        check('theme/light/style.css' not in idx,
              'index: no light theme reference')

              'index: theme bootstrap runs before app.js')
